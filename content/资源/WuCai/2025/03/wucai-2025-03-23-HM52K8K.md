 前文提到日前市面上的多种向量数据库，先针对目前比较火的开源向量数据库milvus做个简单的介绍。Milvus向量数据库最初始在2019年创建，诞生缘由：用于处理大规模的向量搜索和相似性计算，经常在特定的领域：推荐系统、计算机视觉、自然语言处理等；唯一目标是存储，索引和管理由深度神经网络和其他机器学习(ML)模型生成的大规模嵌入向量，即：用来专门存储和检索高维向量数据。同时，Milvus官方参考的中文文档地址：https://www.milvus-io.com。

 截止到2024年4月底，Milvus向量数据库已发布到2.4.0版本，可支持单机版，集群版以及支持GPU版本，从GitHub Release的时间来看，各个版本的发布时间如下： ![](https://g1proxy.wimg.site/sX55mVGGxZSKmFw1mhXTlfHvhhaGs74R-BKJNvrfMajw/https://mmbiz.qpic.cn/sz_mmbiz_png/f1jPRZgKIN20icbxsldPzkPYRs94DEUdBJhCk6Nojeg8LPia7TSmpxfJQ0X9v3QAOniamjXBB3T38A6Zcu1duKxnw/640?wx_fmt=png&from=appmsg)

7.利用Milvus向量数据库搭建简易FAQ Engine


```python
import pandas as pd
from pymilvus import connections, FieldSchema, CollectionSchema, DataType, Collection, utility
from towhee import pipe, ops
import numpy as np
from towhee.datacollection import DataCollection
import csv
import gradio


def create_milvus_collection(collection_name, dim):
    if utility.has_collection(collection_name):
        utility.drop_collection(collection_name)
    fields = [
        FieldSchema(name='id', dtype=DataType.VARCHAR, descrition='ids', max_length=500, is_primary=True,
                    auto_id=False),
        FieldSchema(name='embedding', dtype=DataType.FLOAT_VECTOR, descrition='embedding vectors', dim=dim)
    ]
    schema = CollectionSchema(fields=fields, description='reverse image search')
    collection = Collection(name=collection_name, schema=schema)

    # create IVF_FLAT index for collection.
    index_params = {
        'metric_type': 'L2',
        'index_type': "IVF_FLAT",
        'params': {"nlist": 2048}
    }
    collection.create_index(field_name="embedding", index_params=index_params)
    return collection

def chat(message, history):
    history = history or []
    ans_pipe = (
        pipe.input('question')
            .map('question', 'vec', ops.text_embedding.dpr(model_name="./model/dpr-ctx_encoder-single-nq-base"))
            .map('vec', 'vec', lambda x: x / np.linalg.norm(x, axis=0))
            .map('vec', 'res', ops.ann_search.milvus_client(host='localhost', port='19530', collection_name='wuhan_question_answer', limit=1))
            .map('res', 'answer', lambda x: [id_answer[int(i[0])] for i in x])
            .output('question', 'answer')
    )

    response = ans_pipe(message).get()[1][0]
    history.append((message, response))
    return history, history


df = pd.read_csv('./data/questions.csv')
df.head()
id_answer = df.set_index('id')['answer'].to_dict()
connections.connect(host='localhost', port='19530')

collection = create_milvus_collection('question_answer', 768)
insert_pipe = (
    pipe.input('id', 'question', 'answer')
        .map('question', 'vec', ops.text_embedding.dpr(model_name='./model/dpr-ctx_encoder-single-nq-base'))
        .map('vec', 'vec', lambda x: x / np.linalg.norm(x, axis=0))
        .map(('id', 'vec'), 'insert_status', ops.ann_insert.milvus_client(host='localhost', port='19530', collection_name='question_answer'))
        .output()
)

with open('./data/questions.csv', encoding='utf-8') as f:
    reader = csv.reader(f)
    next(reader)
    for row in reader:
        insert_pipe(*row)

collection.flush()
print('Total number of inserted data is {}.'.format(collection.num_entities))

collection.load()
ans_pipe = (
    pipe.input('question')
        .map('question', 'vec', ops.text_embedding.dpr(model_name="./model/dpr-ctx_encoder-single-nq-base"))
        .map('vec', 'vec', lambda x: x / np.linalg.norm(x, axis=0))
        .map('vec', 'res', ops.ann_search.milvus_client(host='localhost', port='19530', collection_name='wuhan_question_answer', limit=1))
        .map('res', 'answer', lambda x: [id_answer[int(i[0])] for i in x])
        .output('question', 'answer')
)

ans = ans_pipe('螺栓断裂')
ans = DataCollection(ans)
ans.show()
print(ans[0]['question'])
print(ans[0]['answer'])


#构建Engine
collection.load()
chatbot = gradio.Chatbot()
interface = gradio.Interface(
    chat,
    ["text", "state"],
    [chatbot, "state"],
    # allow_screenshot=False,
    allow_flagging="never",
)
interface.launch(inline=True, share=True)
```
